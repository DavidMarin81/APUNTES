# Controller
## Estructura típica de un `Controller` Rest en .NET
~~~C#
[Route("api/[controller]")]
[ApiController]
public class PersonController : ControllerBase
{
    private readonly IPersonRepository _repository;

    public PersonController(IPersonRepository repository)
    {
        _repository = repository;
    }

    // A partir de aquí: GET, POST, PUT, DELETE...
}
~~~

## GET -> Listar datos (`SELECT`)
- Obtener lista completa
    ~~~C#
    [HttpGet]
    public async Task<IActionResult> GetAll()
    {
        var persons = await _repository.GetAllAsync();
        return Ok(persons); // 200
    }
    ~~~
- Obtener por ID
    ~~~C#
    [HttpGet("{id}")]
    public async Task<IActionResult> GetById(int id)
    {
        if (id <= 0)
            return BadRequest("Id inválido");

        var person = await _repository.GetByIdAsync(id);

        if (person == null)
            return NotFound();

        return Ok(person);
    }
    ~~~

## POST -> Crear un registro (`INSERT INTO`)
~~~C#
[HttpPost]
public async Task<IActionResult> Create(Person person)
{
    if (person == null)
        return BadRequest("Datos inválidos");

    if (string.IsNullOrWhiteSpace(person.Name))
        return BadRequest("El nombre es obligatorio");

    var result = await _repository.AddAsync(person);

    if (result > 0)
        return Created($"api/person/{person.Id}", person);

    return StatusCode(500, "Error al insertar en la base de datos");
}
~~~

## PUT -> Actualizar un registro (`UPDATE`)
~~~C#
[HttpPut("{id}")]
public async Task<IActionResult> Update(int id, Person person)
{
    if (id != person.Id)
        return BadRequest("El id no coincide");

    var exists = await _repository.GetByIdAsync(id);
    if (exists == null)
        return NotFound();

    var result = await _repository.UpdateAsync(person);

    return result > 0 ? Ok("Actualizado correctamente") : 
                        StatusCode(500, "Error al actualizar");
}
~~~

## DELETE -> Eliminar un registgro (`DELETE`)
~~~C#
[HttpDelete("{id}")]
public async Task<IActionResult> Delete(int id)
{
    if (id <= 0)
        return BadRequest("Id inválido");

    var exists = await _repository.GetByIdAsync(id);
    if (exists == null)
        return NotFound("No existe el recurso");

    var result = await _repository.DeleteAsync(id);

    return result > 0 ? NoContent() : 
                        StatusCode(500, "Error al eliminar");
}
~~~

## Puede llevar el `[Authorize]`
- Es un atributo que se coloca en `controllers` o métodos para exigir que el usuario esté autenticado antes de poder acceder
    - ✅ Si el usuario tiene sesión o token válido -> puede entrar
    - ❌ Si el usuario no está autenticado -> el servidor responde con **401 Unauthorized**
    ~~~C#
    [Authorize]
    [HttpGet("perfil")]
    public IActionResult GetPerfil()
    {
        return Ok("Este es tu perfil");
    }
    ~~~
